cmake_minimum_required(VERSION 3.16)
project(air_monitor_host LANGUAGES C CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 23)
set(AIR "${CMAKE_CURRENT_SOURCE_DIR}/../../main/boards/custom/waveshare-air-monitor-1.8-v1")
add_library(air_core "${AIR}/sensirion_protocol.cc" "${AIR}/air_model.cc"
                     "${AIR}/sensor_schedule.cc" "${AIR}/air_controls.cc" "${AIR}/air_ui_state.cc"
                     "${AIR}/air_alert_policy.cc" "${AIR}/air_chat_model.cc" "${AIR}/air_solar_theme.cc")
target_include_directories(air_core PUBLIC "${AIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../../main")
target_compile_options(air_core PRIVATE -Wall -Wextra -Werror)
foreach(name protocol model schedule controls display_events shutdown_coordinator alert_policy clock chat_model solar_theme)
    add_executable(test_${name} test_${name}.cc)
    target_link_libraries(test_${name} PRIVATE air_core)
    target_compile_options(test_${name} PRIVATE -Wall -Wextra -Werror)
    add_test(NAME ${name} COMMAND test_${name})
endforeach()
add_executable(test_recovery test_recovery.cc "../../main/storage_init.cc")
target_include_directories(test_recovery PRIVATE "../../main")
target_compile_options(test_recovery PRIVATE -Wall -Wextra -Werror)
add_test(NAME recovery COMMAND test_recovery)
add_executable(test_settings_failure test_settings_failure.cc "../../main/storage_init.cc" "../../main/settings.cc")
target_include_directories(test_settings_failure PRIVATE fakes "../../main")
target_compile_definitions(test_settings_failure PRIVATE ESP_PLATFORM)
target_compile_options(test_settings_failure PRIVATE -Wall -Wextra -Werror)
add_test(NAME settings_failure COMMAND test_settings_failure)
add_executable(test_wifi_recovery test_wifi_recovery.cc "../../main/storage_init.cc")
target_include_directories(test_wifi_recovery PRIVATE network_fakes "../../main")
target_compile_definitions(test_wifi_recovery PRIVATE ESP_PLATFORM CONFIG_USE_HOTSPOT_WIFI_PROVISIONING)
target_compile_options(test_wifi_recovery PRIVATE -Wall -Wextra -Werror)
add_test(NAME wifi_recovery COMMAND test_wifi_recovery)
